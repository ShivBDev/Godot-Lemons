docker run
--name godot-lemon-postgres     // docker convenient name
-e POSTGRES_USER=admin          // master username for db
-e POSTGRES_PASSWORD=[HIDDEN]   // 
-e POSTGRES_DB=godot_lemon_db   // postgres db name
-p 5432:5432                    // port mapping, localPort:dockerPort
-d                              // run detached (frees up terminal)
postgres:18                     // postgres version
